        self.EnterSaleID.clear()
        self.EnterSalePrice.clear()
        self.Customername.clear()
        self.contactnumber_3.clear()

        print("Sales fields cleared")

    def save_sale(self):

        sale_id = self.EnterSaleID.text()
        sale_price = self.EnterSalePrice.text()
        customer = self.Customername.text()
        vehicle = self.vehicleSoldComboBox_4.currentText()
        payment = self.paymentmethodsComboBox.currentText()

        date = self.dateEdit.date().toString(
            "dd/MM/yyyy"
        )

        if sale_id == "" or sale_price == "" or customer == "":

            QtWidgets.QMessageBox.warning(
                None,
                "Missing Information",
                "Please enter the Sale ID, Sale Price and Customer Name."
            )

        else:

            print("Sale Saved")
            print("Sale ID:", sale_id)
            print("Vehicle:", vehicle)
            print("Sale Price:", sale_price)
            print("Customer:", customer)
            print("Payment Method:", payment)
            print("Sale Date:", date)

            self.contactnumber_3.setText(
                "Sales Summary\n"
                "Sale ID: " + sale_id + "\n"
                "Vehicle: " + vehicle + "\n"
                "Customer: " + customer + "\n"
                "Sale Price: R" + sale_price + "\n"
                "Payment Method: " + payment + "\n"
                "Sale Date: " + date
            )

            QtWidgets.QMessageBox.information(
                None,
                "Sale Saved",
                "Sale saved successfully."
            )
